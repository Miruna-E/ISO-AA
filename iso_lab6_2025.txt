sunt utilizate argumentele pentru comenzi sau pentru programele compilate.

ex:
hello.sh domnule Grigore

echo $1 $2
//=> Buna ziua domnule Grigore

$# = nr de argumente cu care a fost apelat un script

if [ $# -ne 2 ]
then
 echo "Prea puține argumente!"

else

fi

#!/bin/bash
let cost_total=0
while read line ; do
        if [[ $( echo "$line" | wc -w ) -eq 4 ]] ; then
                let cost_total=cost_total+10
        fi
        suma_crt=$( echo "$line" | awk '{print $NF}' )
        suma_crt=${suma_crt::-1}
        let cost_total=$((cost_total + suma_crt))
done < baza_date
echo $cost_total

#!/bin/bash
addresses=(google.com wikipedia.org 192.168.145.236)
for element in "${addresses[@]}" ; do
        ping -c 1 $element
        if [[ $? -eq 0 ]] ; then
                echo "$(date) - $element is reachable" >> network_check
        else
                echo "$(date) - $element is unreachable" >> network_check
        fi
done

./verifica_retea
