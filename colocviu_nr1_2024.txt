#!/bin/bash
echo "Se va calcula suma a doua numere citite, ambele < 20"
read a
while [[ ${a} -ge 20 ]] ; do
        read a
done
read b
while [[ ${b} -ge 20 ]] ; do
        read b
done
let suma=0
for (( i=${a}+1; i < ${b}; i++ )) ; do
        let suma+=${i}
done
echo "Suma numerelor din (${a}; ${b}) este " ${suma}

./sum
